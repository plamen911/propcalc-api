# rsync exclude list for deploying this app to the icnhost production host.
# Read by ./deploy.sh.
#
# Pattern semantics (rsync/openrsync):
#   - a LEADING slash anchors the pattern to the transfer root
#   - a pattern with no slash matches a name at ANY depth
# The app-level directories below MUST stay anchored: a bare `docs/` would also
# swallow public/docs/ (tracked PDFs the API serves), and a bare `.htaccess`
# would match at every depth.

# --- deliberately not uploaded / server-owned runtime -----------------------
# Dependencies are installed on the server instead (composer.phar is there).

/var/
/.env

/.env.local.php
/.env.*.local
# JWT keys are generated per environment and live only on the server.
/config/jwt/
# Generated on the server by `assets:install` (composer auto-scripts).
/public/bundles/

# --- server-owned: managed by cPanel/the host, never overwrite --------------
/.htaccess
/php*.fcgi
/php*-fcgi.ini
/cgi-bin/
/.well-known/
/error_log
/php_errorlog
/.ftpquota
# May carry cPanel PHP-handler lines on prod; the tracked copy is for local dev.
public/.htaccess
public/.well-known/
public/error_log
